## Proposal: RAG-NIMBF

### Design
Add a `llm:` provider to the reranker factory implementing listwise reranking: a small local LLM (default `ollama:qwen3:0.6b`) reads the query + numbered candidate list and emits a relevance-ordered permutation. Plugs into the existing `Reranker` protocol (`rerank(query, documents, top_k)`), so adapters, `retriever.py` and the matrix runner need no changes — only the factory dispatch in `get_reranker`. Defaults: temperature 0, single listwise pass, graceful fallback to input order on parse failure (scored as identity permutation).

Data flow: `config.reranker_model="llm:qwen3:0.6b"` → `get_reranker` → `ListwiseLLMReranker` → `get_chat_model` (existing provider factory) → permutation → reordered docs.

"""Draft generated by RAG Radar — untested, review before applying.

Listwise LLM reranker (Re3Rank-style): a small instruction-tuned LLM
re-orders the candidate list in one pass instead of scoring pairs.
"""
from __future__ import annotations

import re

from langchain_core.documents import Document

from benchmark.providers import get_chat_model
from benchmark.tracing import observe

_PROMPT = (
    "Rank the following passages by relevance to the query.\n"
    "Query: {query}\n\n"
    "{docs}\n\n"
    "Output ONLY the passage numbers, best first, separated by commas "
    "(e.g. 3,1,4,2). Include every number exactly once."
)


class ListwiseLLMReranker:
    """Reranker that asks a local LLM for a permutation of the candidates."""

    def __init__(self, model_name: str, base_url: str = "http://localhost:11434") -> None:
        provider, name = model_name.split(":", 1) if ":" in model_name else ("ollama", model_name)
        self.model = get_chat_model(
            provider=provider, model_name=name, base_url=base_url,
            max_tokens=128, temperature=0.0,
        )

    def _parse(self, text: str, n: int) -> list[int]:
        seen: list[int] = []
        for tok in re.findall(r"\d+", text or ""):
            i = int(tok)
            if 0 < i <= n and i not in seen:
                seen.append(i)
        seen += [i for i in range(1, n + 1) if i not in seen]
        return seen[:n]

    @observe(name="rerank")
    def rerank(
        self, query: str, documents: list[Document], top_k: int
    ) -> list[Document]:
        if not documents:
            return []
        docs_block = "\n".join(
            f"[{i}] {d.page_content[:500]}" for i, d in enumerate(documents, 1)
        )
        try:
            out = self.model.invoke(_PROMPT.format(query=query, docs=docs_block))
            order = self._parse(getattr(out, "content", str(out)), len(documents))
        except Exception:
            order = list(range(1, len(documents) + 1))
        return [documents[i - 1] for i in order[:top_k]]
```

# Draft generated by RAG Radar — untested, review before applying.
"""Optional reranking stage: re-orders retrieved docs with a cross-encoder."""
import os
import threading
from typing import Any, List

from langchain_core.callbacks import CallbackManagerForRetrieverRun
from langchain_core.documents import Document
from langchain_core.retrievers import BaseRetriever

_lock = threading.Lock()
_model: Any = None


def _get_model():
    global _model
    if _model is None:
        from sentence_transformers import CrossEncoder
        _model = CrossEncoder(os.environ["RERANKER_MODEL"])
    return _model


class RerankingRetriever(BaseRetriever):
    """Wraps another retriever and re-scores its documents."""

    top_k: int = 5

    def _get_relevant_documents(
        self, query: str, *, run_manager: CallbackManagerForRetrieverRun
    ) -> List[Document]:
        docs = self.unwrap.invoke(query)  # type: ignore[attr-defined]
        if not docs:
            return []
        with _lock:
            scores = _get_model().predict(
                [(query, d.page_content) for d in docs]
            )
        ranked = sorted(zip(scores, docs), key=lambda x: x[0], reverse=True)
        return [d for _, d in ranked[: self.top_k]]

    # small helper so the wrapper stays generic
    def __init__(self, **kwargs: Any) -> None:
        super().__init__(**kwargs)


def maybe_wrap(retriever, top_k: int = 5):
    """Return a reranking wrapper if RERANKER_MODEL is set, else the input."""
    model_id = os.environ.get("RERANKER_MODEL", "").strip()
    if not model_id:
        return retriever
    wrapper = RerankingRetriever(top_k=top_k)
    wrapper.unwrap = retriever
    return wrapper
```
