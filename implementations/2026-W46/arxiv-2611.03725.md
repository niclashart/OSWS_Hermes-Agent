## Proposal: RAG-NIMBF

### Design
Add a lightweight, reference-free fault-localization post-processor over the traces NIMBF already records. Every stage span (`rerank`, `retrieve`, `generate`, …) already carries attributes via `benchmark/tracing.py` (`@observe`); new module `benchmark/fault_localization.py` reads the run's span list from MLflow and applies rule-based checks (empty retrieval, zero candidates after threshold, refusal answer, generation timeout) + optional LLM judge on the transcript, then emits per-stage fault labels and a `stage_fault_f1` when gold fault labels are present (e.g. injected-fault datasets). No new axis value: it is a diagnostic report attached to each run, plus an optional `fault_localization_enabled` config flag.

Data flow: run finishes → `fault_localization.analyze_run(run_id)` pulls spans → rules + optional judge → `faults.json` next to results + MLflow tags `fault_stage`, `fault_confidence`.

#### NEW `benchmark/fault_localization.py`
```python
"""Draft generated by RAG Radar — untested, review before applying.

Reference-free per-stage fault localization for benchmark runs.
Rules operate on stage spans recorded by benchmark.tracing.
"""
from __future__ import annotations

import json
import os
from dataclasses import dataclass, field, asdict
from pathlib import Path

STAGES = ("chunker", "embedder", "retriever", "reranker", "llm")


@dataclass
class Fault:
    stage: str
    rule: str
    confidence: float
    detail: str = ""


@dataclass
class RunDiagnosis:
    run_id: str
    faults: list[Fault] = field(default_factory=list)

    @property
    def faulty_stages(self) -> list[str]:
        return sorted({f.stage for f in self.faults})


def _rule_empty_retrieval(spans: list[dict]) -> list[Fault]:
    out = []
    for s in spans:
        if s.get("name") == "retrieve" and s.get("attributes", {}).get("num_docs", 1) == 0:
            out.append(Fault("retriever", "empty_retrieval", 0.9))
    return out


def _rule_all_filtered(spans: list[dict]) -> list[Fault]:
    out = []
    for s in spans:
        a = s.get("attributes", {})
        if s.get("name") == "retrieve" and 0 < a.get("num_docs", 1) < a.get("num_candidates", 1):
            out.append(Fault("retriever", "over_filtering", 0.5,
                             f"{a.get('num_docs')}/{a.get('num_candidates')} docs survived"))
    return out


def _rule_refusal_answer(spans: list[dict]) -> list[Fault]:
    from benchmark.custom_metrics import is_refusal_answer
    out = []
    for s in spans:
        if s.get("name") == "generate":
            ans = s.get("attributes", {}).get("answer", "")
            if ans and is_refusal_answer(ans):
                out.append(Fault("llm", "refusal_answer", 0.6, ans[:120]))
    return out


def _rule_stage_latency(spans: list[dict], threshold_s: float = 60.0) -> list[Fault]:
    out = []
    for s in spans:
        dur = s.get("duration_s") or 0.0
        if dur > threshold_s:
            out.append(Fault(s.get("name", "?"), "latency_outlier", 0.4, f"{dur:.1f}s"))
    return out


RULES = [_rule_empty_retrieval, _rule_all_filtered, _rule_refusal_answer, _rule_stage_latency]


def analyze_spans(run_id: str, spans: list[dict]) -> RunDiagnosis:
    d = RunDiagnosis(run_id=run_id)
    for rule in RULES:
        d.faults.extend(rule(spans))
    return d


def analyze_run(run_id: str, output_dir: str = "results") -> RunDiagnosis:
    """Pull spans from MLflow for *run_id* and write faults.json."""
    import mlflow
    client = mlflow.tracking.MlflowClient()
    spans = []
    try:
        for trace in client.search_traces(run_ids=[run_id]):
            for span in trace.spans:
                spans.append({
                    "name": span.name,
                    "duration_s": getattr(span, "duration_ms", 0) / 1000.0,
                    "attributes": dict(getattr(span, "attributes", {}) or {}),
                })
    except Exception:
        pass
    d = analyze_spans(run_id, spans)
    p = Path(output_dir) / run_id / "faults.json"
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(asdict(d), indent=2))
    return d


def score_against_gold(diagnosis: RunDiagnosis, gold_stages: list[str]) -> dict:
    """Pooled stage-level precision/recall/F1 against injected-fault labels."""
    pred = set(diagnosis.faulty_stages)
    gold = set(gold_stages)
    tp = len(pred & gold)
    fp, fn = len(pred - gold), len(gold - pred)
    p = tp / (tp + fp) if tp + fp else 0.0
    r = tp / (tp + fn) if tp + fn else 0.0
    f1 = 2 * p * r / (p + r) if p + r else 0.0
    return {"precision": p, "recall": r, "f1": f1}
```

```python
# Draft generated by RAG Radar — untested, review before applying.
"""Per-request stage records + rule-based fault flags (RAGLog-style)."""
import threading
import time
from collections import deque

_lock = threading.Lock()
_buffer: deque = deque(maxlen=100)


def record_stage(stage: str, duration_s: float, **attrs) -> None:
    with _lock:
        _buffer.append({
            "ts": time.time(), "stage": stage,
            "duration_s": duration_s, **attrs,
        })


def diagnose(last_n: int = 20) -> list[dict]:
    with _lock:
        records = list(_buffer)[-last_n:]
    faults = []
    for r in records:
        if r["stage"] == "retrieve" and r.get("num_docs", 1) == 0:
            faults.append({"stage": "retriever", "rule": "empty_retrieval",
                           "detail": r.get("question", "")[:80]})
        if r["stage"] == "generate" and r.get("refusal"):
            faults.append({"stage": "llm", "rule": "refusal_answer",
                           "detail": r.get("answer", "")[:80]})
        if r["duration_s"] > 30.0:
            faults.append({"stage": r["stage"], "rule": "latency_outlier",
                           "detail": f"{r['duration_s']:.1f}s"})
    return faults
```
