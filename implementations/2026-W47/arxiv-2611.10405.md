## Proposal: Enterprise_RAG_Blueprint

### Design
Add an optional, default-off reranking stage on top of the existing (hybrid) retriever, based on the branch that already carries related retrieval work (`feat/mcp-integration`, which contains `feat/hybrid-retrieval`):

- New module `chain/reranker.py`: `RerankerConfig` (env-driven: `RERANKER_ENABLED`, `RERANKER_MODEL`, `RERANKER_TOP_K`, `RERANKER_CANDIDATE_K`) and `create_reranker()` returning `None` when disabled — mirroring the repo's env-config style (`chain/handle_llms.py:9-12`).
- `OptionalRerankerRetriever`, a thin LangChain `BaseRetriever` wrapper: fetch `RERANKER_CANDIDATE_K` (default 20) docs from the wrapped retriever, score `(query, doc)` pairs with `sentence_transformers.CrossEncoder(RERANKER_MODEL)`, return the top `RERANKER_TOP_K` (default 5 — matches `handle_llms.call_llm`'s `returned_docs=5`). Because it is a `BaseRetriever`, `load_chain` (retriever branch, `load_chain.py:82-90`) and `rag_chain` (`load_chain.py:143`) work unchanged, including role filtering, which happens inside the wrapped retriever.
- Failure containment: on any reranker exception, log a warning and pass through the wrapped retriever's documents (production must not break because a model download failed).
- The hybrid branch (`chain/hybrid_retriever.py` on the same base) composes naturally: hybrid retriever → reranker wrapper → chain.

#### NEW `chain/reranker.py`
```python
"""Draft generated by RAG Radar — untested, review before applying.

Optional cross-encoder reranking stage (Yuanbao-Reranker-style pointwise
scorer, works with any sentence-transformers CrossEncoder).
"""
import logging
import os
from typing import List

from langchain_core.callbacks import CallbackManagerForRetrieverRun
from langchain_core.documents import Document
from langchain_core.retrievers import BaseRetriever

logger = logging.getLogger(__name__)

RERANKER_ENABLED = os.getenv("RERANKER_ENABLED", "").lower() in ("1", "true", "yes")
RERANKER_MODEL = os.getenv("RERANKER_MODEL", "tencent/Yuanbao-Reranker-0.6B")
RERANKER_TOP_K = int(os.getenv("RERANKER_TOP_K", "5"))
RERANKER_CANDIDATE_K = int(os.getenv("RERANKER_CANDIDATE_K", "20"))


def create_reranker():
    """Return a CrossEncoder instance, or None when reranking is disabled."""
    if not RERANKER_ENABLED:
        return None
    from sentence_transformers import CrossEncoder

    logger.info("Loading reranker model %s", RERANKER_MODEL)
    return CrossEncoder(RERANKER_MODEL)


class OptionalRerankerRetriever(BaseRetriever):
    """Wrap a retriever and rerank its candidates with a cross-encoder."""

    base_retriever: BaseRetriever
    reranker: object  # CrossEncoder | None (None = pass-through)
    top_k: int = RERANKER_TOP_K
    candidate_k: int = RERANKER_CANDIDATE_K

    def _get_relevant_documents(
        self, query: str, *, run_manager: CallbackManagerForRetrieverRun
    ) -> List[Document]:
        docs = self.base_retriever.invoke(query)[: self.candidate_k]
        if self.reranker is None or not docs:
            return docs[: self.top_k]
        try:
            pairs = [(query, doc.page_content) for doc in docs]
            scores = self.reranker.predict(pairs)
            ranked = sorted(zip(scores, docs), key=lambda x: x[0], reverse=True)
            return [doc for _, doc in ranked[: self.top_k]]
        except Exception as e:  # never break the chain on reranker failure
            logger.warning("Reranking failed, passing through base results: %s", e)
            return docs[: self.top_k]


def wrap_retriever(base_retriever: BaseRetriever) -> BaseRetriever:
    """Wrap with reranking when enabled; otherwise return the retriever unchanged."""
    if not RERANKER_ENABLED:
        return base_retriever
    return OptionalRerankerRetriever(
        base_retriever=base_retriever, reranker=create_reranker()
    )
```

```python
"""Draft generated by RAG Radar — untested, review before applying."""
import os
import sys
from unittest.mock import MagicMock

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import reranker
from reranker import OptionalRerankerRetriever, wrap_retriever


def _doc(i, text="text"):
    return MagicMock(page_content=f"{text} {i}", metadata={"source": f"d{i}.pdf"})


class _FakeCrossEncoder:
    def __init__(self, scores):
        self.scores = scores

    def predict(self, pairs):
        return self.scores[: len(pairs)]


def test_disabled_passthrough(monkeypatch):
    monkeypatch.setattr(reranker, "RERANKER_ENABLED", False)
    base = MagicMock()
    assert wrap_retriever(base) is base


def test_rerank_orders_by_score():
    base = MagicMock()
    docs = [_doc(1), _doc(2), _doc(3)]
    base.invoke.return_value = docs
    rr = OptionalRerankerRetriever(
        base_retriever=base, reranker=_FakeCrossEncoder([0.1, 0.9, 0.5]), top_k=2
    )
    out = rr.invoke("q")
    assert len(out) == 2
    assert out[0].page_content == "text 2"  # highest score first


def test_reranker_failure_falls_back():
    base = MagicMock()
    docs = [_doc(1), _doc(2)]
    base.invoke.return_value = docs
    bad = MagicMock()
    bad.predict.side_effect = RuntimeError("boom")
    rr = OptionalRerankerRetriever(base_retriever=base, reranker=bad, top_k=1)
    assert rr.invoke("q")[0].page_content == "text 1"


def test_candidate_k_limits_base_calls():
    base = MagicMock()
    docs = [_doc(i) for i in range(10)]
    base.invoke.return_value = docs
    rr = OptionalRerankerRetriever(
        base_retriever=base, reranker=_FakeCrossEncoder([0.5] * 10),
        top_k=2, candidate_k=3,
    )
    assert len(rr.invoke("q")) == 2
```
